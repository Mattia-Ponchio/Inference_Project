\section{Introduction}

\begin{frame}{No U-Turn Sampler}
	\begin{center}
		 
		\emph{HMC} is an \emph{MCMC} algorithm that avoids random walk behavior and sensitivity to correlated parameters.\\ %, by taking a series of steps informed by first-order gradient information. \\
		However, it requires the tuning of two user-specified parameters: step size $\epsilon$ and number of steps $L$. \\ 
		
		\pause
		%\vspace{1pt}
		\(\Downarrow\)
		\vspace{1pt}\\
		The \emph{No-U-Turn Sampler} (NUTS) is an extension to HMC that eliminates the need to set a number of steps $L$.
		
		\pause
		%\vspace{1pt}
		\(\Downarrow\)
		\vspace{1pt}\\
		\emph{NUTS} uses a recursive algorithm to build a set of candidate points, stopping automatically when it starts retrace its steps. 
		
		\pause
		\begin{tikzpicture}
			\node[opacity=0.5] {
				\includegraphics[width=0.6\textwidth]{images/immagini_leonardo/NUTS_diagram.png}
			};
		\end{tikzpicture}
		
	\end{center}
	
\end{frame}

\begin{frame}{Hamiltonian Monte Carlo}
	HMC generates proposals with high acceptance probability by suppressing the random walk behavior, but it comes with a price:
	\pause
	\begin{itemize}
		\item Comuputing the gradient of the log-posterior at each step. \\
		\pause
		\item Tuning two user-specified parameters: step size $\epsilon$ and number of steps $L$.
		Tipically, $\epsilon$ can be tuned during warm-up, but $L$ can require costly tuning runs. \\
	\end{itemize}
	\pause
	\centering

	\begin{tikzpicture}
			\node[opacity=0.75] {
				    \includegraphics[width=0.75\textwidth]{images/immagini_leonardo/HMC.png}
			};
		\end{tikzpicture}

\end{frame}

%----------------------------------------------------------------------------------------------------

\begin{frame}{No U-Turn HMC}
	We want a sampler able to retain the HMC suppression on random walk behavior, but without the need to tune the number of Leap Frog steps $L$. \\
	\pause
	\begin{itemize}
		\item Starting from a point in the parameter space $\theta$, the algorithm builds a set of candidate points, doubling the tree size at each iteration. \\
		\pause
		\item To detect when to stop, the algorithm checks if the trajectory starts to turn back on itself: given a proposed point $\tilde{\theta}$, the algorithm checks if the dot product of the momentum vectors at $\theta$ and $\tilde{\theta}$ is negative. \\
		$(\theta - \tilde{\theta}) \cdot \frac{d}{dt}(\theta - \tilde{\theta}) =(\theta - \tilde{\theta}) \cdot \tilde{r} < 0$ \\
		\pause
		\item To preserve time reversibility, NUTS runs the Hamiltonian dynamics in both forward and backward directions. \\
	\end{itemize}
	

\end{frame}

\section{NUTS algorithm}

\begin{frame}{Probability slicing}
	\begin{center}
		 
		To simplify the derivation and implementation of the algorithm, NUTS uses a slice variable $u$ with conditional distribution $p(u|\theta, r) = \mathcal{U}(u; [0, exp{\mathcal{L}(\theta)-\frac{1}{2}r\cdot r}]) $. \\ 
		This way $p(\theta, r | u ) = \mathcal{U}(\theta, r; {\theta', r' | exp{\mathcal{L}(\theta)-\frac{1}{2}r\cdot r} \geq u})$. \\
		\pause
		\begin{tikzpicture}

			};
		\end{tikzpicture}
		
	\end{center}
	
\end{frame}

\begin{frame}{High-level NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.5\textwidth}
        \small
		
			\begin{itemize}
				\item The algorithm resamples $u|\theta, r$ at each iteration,
				\pause
				\item uses Leapfrog integrator to trace a path forward and backwards first 1 step,
				\pause
				\item then doubling for successive iterations.
				\pause
				\item This doubling process implicitly builds a balanced binary tree whose leaf nodes correspond to position-momentum states.
				\pause
				\item The doubling is halted when the subtrajectory starts to double back on itself.
			\end{itemize}

		\end{column}
		
		\begin{column}<1->{0.5\textwidth}
		\onslide<1->
            \centering
			\begin{tikzpicture}

				};
			\end{tikzpicture}
			\small 
			Example of a NUTS trajectory, with the slice variable $u$.
        \end{column}    
    \end{columns}

\end{frame}

\begin{frame}{Detailed NUTS algorithm}
	
	
	\begin{columns}

        \begin{column}{0.53\textwidth}
        \small
		
		Given $\mathcal{B}$, set of position-momentum states traced by the leapfrog integrator, we pick deterministicly $\mathcal{C}$, the set of candidate points.
		\pause
			\begin{itemize}
				\item Transitions between states in $\mathcal{C}$ must preserve proability volume and satisfy detailed balance.
				\pause
				\item The starting position-momentum state $(\theta, r)$ is added to $\mathcal{C}$.
				\pause
				\item Each point in $\mathcal{C}$ must satisfy the slice variable condition $exp{\mathcal{L}(\theta)-\frac{1}{2}r\cdot r} \geq u$.
				\pause
				\item Each point in $\mathcal{C}$ must have the same probability of constructing the trajectory $\mathcal{B}$ and candidate set $\mathcal{C}$.
			\end{itemize}

		\end{column}

\end{frame}

\begin{frame}{Detailed NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.53\textwidth}
        \small

		We can ensure that with the following procedure at iteration $t$:
		\pause
			\begin{itemize}
				\item Sample $r \sim \mathcal{N}(0, I)$,
				\pause	
				\item $u \sim \mathcal{U}(0, exp{\mathcal{L}(\theta^t)-\frac{1}{2}r\cdot r})$.
				\pause
				\item Sample $\mathcal{B}, \mathcal{C} \sim p(\mathcal{B}, \mathcal{C} | \theta^t, r, u, \epsilon)$.
				\pause
				\item Sample $\theta^{t+1}, r \sim T(\theta^t, r, \mathcal{C})$, transition kernel.
			\end{itemize}
		\pause
		Where the transition kernel $T(\theta, r, \mathcal{C})$ has leave the distribution over $\mathcal{C}$ invariant, valid because $\epsilon$ is uniform on the elements of $\mathcal{C}$ and $p(\theta, r | u, \mathcal{B}, \mathcal{C}, \epsilon) \propto \mathbb{I}[(\theta, r)\in \mathcal{C}] $.

		\end{column}
		
        \begin{column}<1->{0.47\textwidth}
		\onslide<1->

\end{frame}

\begin{frame}{Detailed NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.53\textwidth}
        \small

		The trajectory $\mathcal{B}$ is built with a doubling binary tree structure:
		\pause
			\begin{itemize}
				\item First a single node for the starting state.
				\pause
				\item Each iteration $j$ preoceeds with in a direction $v_j \sim \mathcal{U}({-1, 1})$ taking $2^j$ Leapfrog steps.
				\pause
				\item The iteration stops when the trajectory starts to double back on itself
				\pause
				\item or when the error in the simulation becomes too large, $\mathcal{L}(\theta)-\frac{1}{2}r\cdot r - \log{u} < -\Delta_{max}$.
			\end{itemize}
			\pause
			The doubling process defines the distribution $p(\mathcal{B} | \theta^t, r, u, \epsilon)$.

		\end{column}
		
        \begin{column}<1->{0.47\textwidth}

\begin{frame}{Detailed NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.55\textwidth}
        \small

			\begin{itemize}
				\item NUTS algorithm repeatedly calls the BuildTree function, returning left and right endpoints of the trajectory, and all visited states $\mathcal{C'}$.
				\pause
				\item If one of the two stepping conditions is met, the algorithm terminates and the new candidates $\mathcal{C'}$ are rejected.
				\pause
				\item Otherwise, the new candidates $\mathcal{C'}$ are added to the set of candidate points $\mathcal{C}$.
				\pause
				\item If a global U turn condition is met, the doubling process is terminated.
				\pause
				\item The new position-momentum state $(\theta^{t+1}, r)$ is sampled from the set of candidate points $\mathcal{C}$.
			\end{itemize}
			

		\end{column}

\begin{frame}{Optimized NUTS algorithm}
	
	\begin{columns}

		
        \begin{column}{0.5\textwidth}
        \small
		
		Some optimizations can be made to the NUTS algorithm:
		\pause
			\begin{itemize}
				\item Early stopping when a stopping criterion is met mid-process.
				\pause
				\item Single Proposed State kept in memory, by sampling a candidate for each subtree and drawing from them using their relative weights (i.e. the size of the subtree).
				\pause
				\item Adaptive tuning of the step size parameter $\epsilon$, using dual averaging.

\begin{frame}{Dual Averaging for $\epsilon$}
	

	\small
	$\epsilon$ is set using stochastic optimization with vanishing adaptation.
	\pause

	\begin{columns}

		
        \begin{column}{0.5\textwidth}
        \small
		
		
			\begin{itemize}
				\item Defining an expectation $h(\epsilon) = \mathbb{E}_{t}[H_t |x] = \lim_{T \to \infty} \frac{1}{T} \sum_{t=1}^T \mathbb{E}_{t}[H_t | x]$.
				\pause
				\item Using the desired acceptance probability $\delta$ to define the error $H_t = \delta - \alpha_t$, where $\alpha_t$ is the acceptance probability at iteration $t$.
				\pause
				\item Updating the parameter with the statistics $H_t$: $x_{t+1} \leftarrow \mu - \frac{\sqrt{t}}{\gamma}\frac{1}{t +t_0}\sum_{i=1}^{t}H_i$, $\bar{x}_{t+1} \leftarrow \bar{x}_t + \eta_t x_{t+1} + (1 - \eta_t )\bar{x}_t$ with $\eta_t = t^{-k}$. 
			\end{itemize}
			

		\end{column}
		
        \begin{column}<1->{0.5\textwidth}
		\onslide<1->
            \centering
			\begin{tikzpicture}
				\node[opacity=0.75] {
            		\includegraphics[width=1.1\textwidth]{images/immagini_leonardo/Dual_Averaging.png} 
				};
			\end{tikzpicture} \\
        \end{column}    
    
	
	\end{columns}
	\pause
	\small
	\vspace{1pt}
	$\mu=\log{10\epsilon_1}$ is a freely chosen point, $\gamma = 0.05$ is a free parameter for the speed of convergence, and $t_0=10$ is a free parameter that stabilizes the initial iterations and $k=0.75 \in (0.5, 1]$.

\end{frame}

%----------------------------------------------------------------------------------------------------

\begin{frame}{Heuristic function for $\epsilon$}
	

	\small
	Convergence of Dual Averaging for $\epsilon$ is faster starting from a reasonable setting of parameters.
	\pause

	\begin{columns}

		
        \begin{column}{0.5\textwidth}
        \small
		
		
			\begin{itemize}
				\item This heuristic doubles or halves the value of $\epsilon_1$ until the acceptance probability of the Langevin (Leapfrog) proposal crosses 0.5.
				\pause
				\item NUTS does not have a single acceptance probability, so we use an average HMC acceptance probability over the final doubling iteration states.
			\end{itemize}
			

		\end{column}
		
        \begin{column}<1->{0.55\textwidth}
		\onslide<1->
            \centering
			\begin{tikzpicture}
				\node[opacity=0.75] {
            		\includegraphics[width=1\textwidth]{images/immagini_leonardo/Reasonable_Epsilon.png} 
				};
			\end{tikzpicture} \\
			\pause
			\vspace{3pt}
			$H^{NUTS}=\frac{1}{|\mathcal{B}_t^{f}|}\sum_{\theta, r \in \mathcal{B}_t^{f}} \min{1, \frac{\pi(\theta, r)}{\pi(\theta_{t-1}, r_{t}^{0})}}$.
		\end{column}    
    
	
	\end{columns}
	\small
	\vspace{3pt}
	\pause
	We can use Dual Averaging with $H_t = \delta - H^{NUTS}$ and $x_t = \log{\epsilon_t}$ to coerce the average of $H^{NUTS}$ to $\delta$. \\
	\pause
	\vspace{3pt}
	Now the algorithm only requires an acceptance probability $\delta$ as input, and a number of iterations $M^{adapt}$ for the adaptation phase.

\end{frame}
