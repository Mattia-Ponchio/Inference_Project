\section{Introduction}

\begin{frame}{No U-Turn Sampler}
	\begin{center}
		 
		\emph{HMC} is an \emph{MCMC} algorithm that avoids random walk behaviour and sensitivity to correlated parameters.\\ %, by taking a series of steps informed by first-order gradient information. \\
		However, it requires the tuning of two user-specified parameters: step size $\epsilon$ and number of steps $L$. \\ 
		
		\pause
		%\vspace{1pt}
		\(\Downarrow\)
		\vspace{1pt}\\
		The \emph{No-U-Turn Sampler} (NUTS) is an extension to HMC that eliminates the need to set a number of steps $L$.
		
		\pause
		%\vspace{1pt}
		\(\Downarrow\)
		\vspace{1pt}\\
		\emph{NUTS} uses a recursive algorithm to build a set of candidate points, stopping automatically when it starts retrace its steps. 
		
		\pause
		\begin{tikzpicture}
			\node[opacity=0.5] {
				\includegraphics[width=0.6\textwidth]{images/immagini_leonardo/NUTS_diagram.png}
			};
		\end{tikzpicture}
		
	\end{center}
	
\end{frame}

%----------------------------------------------------------------------------------------------------

\begin{frame}{Hamiltonian Monte Carlo}
	HMC generates proposals with high acceptance probability by suppressing the random walk behaviour, but it comes with a price:
	\pause
	\begin{itemize}
		\item Computing the gradient of the log-posterior at each step. \\
		\pause
		\item Tuning two user-specified parameters: step size $\epsilon$ and number of steps $L$.
		Tipically, $\epsilon$ can be tuned during warm-up, but $L$ can require costly tuning runs. \\
	\end{itemize}
	\pause
	\centering

	\begin{tikzpicture}
			\node[opacity=0.75] {
				    \includegraphics[width=0.75\textwidth]{images/immagini_leonardo/HMC.png}
			};
		\end{tikzpicture}

\end{frame}

%----------------------------------------------------------------------------------------------------

\begin{frame}{No U-Turn HMC}
	We want a sampler able to retain the HMC suppression on random walk behaviour, but without the need to tune the number of Leap Frog steps $L$. \\
	\pause
	\begin{itemize}
		\item Starting from a point in the parameter space $\theta$, the algorithm builds a set of candidate points, doubling the tree size at each iteration. \\
		\pause
		\item To detect when to stop, the algorithm checks if the trajectory starts to turn back on itself: given a proposed point $\tilde{\theta}$, the algorithm checks if the dot product of the momentum vectors at $\theta$ and $\tilde{\theta}$ is negative. \\
		$(\theta - \tilde{\theta}) \cdot \frac{d}{dt}(\theta - \tilde{\theta}) =(\theta - \tilde{\theta}) \cdot \tilde{r} < 0$ \\
		\pause
		\item To preserve time reversibility, NUTS runs the Hamiltonian dynamics in both forward and backward directions. \\
	\end{itemize}
	

\end{frame}

\section{NUTS algorithm}

\begin{frame}{Probability slicing}
	\begin{center}
		 
		To simplify the derivation and implementation of the algorithm, NUTS uses a slice variable $u$ with conditional distribution $p(u|\theta, r) = \mathcal{U}(u; [0, exp{(\mathcal{L}(\theta)-\frac{1}{2}r\cdot r)}]) $. \\ 
		This way $p(\theta, r | u ) = \mathcal{U}(\theta, r; {\theta', r' | \exp{(\mathcal{L}(\theta)-\frac{1}{2}r\cdot r)} \geq u})$. \\

	\end{center}
	
\end{frame}

\begin{frame}{High-level NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.5\textwidth}
        \small
		
			\begin{itemize}
				\item The algorithm resamples $u|\theta, r$ at each iteration,
				\pause
				\item uses Leapfrog integrator to trace a path forward and backwards first 1 step,
				\pause
				\item then doubling for successive iterations.
				\pause
				\item This doubling process implicitly builds a balanced binary tree whose leaf nodes correspond to position-momentum states.
				\pause
				\item The doubling is halted when the subtrajectory starts to double back on itself.
			\end{itemize}

		\end{column}

\begin{frame}{Detailed NUTS algorithm}
	
	
	\begin{columns}

        \begin{column}{0.53\textwidth}
        \small
		
		Given $\mathcal{B}$, set of position-momentum states traced by the leapfrog integrator, we pick deterministically $\mathcal{C}$, the set of candidate points.
		\pause
			\begin{itemize}
				\item Transitions between states in $\mathcal{C}$ must preserve proability volume and satisfy detailed balance.
				\pause
				\item The starting position-momentum state $(\theta, r)$ is added to $\mathcal{C}$.
				\pause
				\item Each point in $\mathcal{C}$ must satisfy the slice variable condition $exp{\mathcal{L}(\theta)-\frac{1}{2}r\cdot r} \geq u$.
				\pause
				\item Each point in $\mathcal{C}$ must have the same probability of constructing the trajectory $\mathcal{B}$ and candidate set $\mathcal{C}$.
			\end{itemize}

		\end{column}

\end{frame}

\begin{frame}{Detailed NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.53\textwidth}
        \small

		We can ensure that with the following procedure at iteration $t$:
		\pause
			\begin{itemize}
				\item Sample $r \sim \mathcal{N}(0, I)$,
				\pause	
				\item $u \sim \mathcal{U}(0, exp{\mathcal{L}(\theta^t)-\frac{1}{2}r\cdot r})$.
				\pause
				\item Sample $\mathcal{B}, \mathcal{C} \sim p(\mathcal{B}, \mathcal{C} | \theta^t, r, u, \epsilon)$.
				\pause
				\item Sample $\theta^{t+1}, r \sim T(\theta^t, r, \mathcal{C})$, transition kernel.
			\end{itemize}
		\pause
		Where the transition kernel $T(\theta, r, \mathcal{C})$ has left the distribution over $\mathcal{C}$ invariant, valid because $\epsilon$ is uniform on the elements of $\mathcal{C}$ and $p(\theta, r | u, \mathcal{B}, \mathcal{C}, \epsilon) \propto \mathbb{I}[(\theta, r)\in \mathcal{C}] $.

		\end{column}
		
        \begin{column}<1->{0.47\textwidth}
		\onslide<1->

\end{frame}

\begin{frame}{Detailed NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.53\textwidth}
        \small

		The trajectory $\mathcal{B}$ is built with a doubling binary tree structure:
		\pause
			\begin{itemize}
				\item First a single node for the starting state.
				\pause
				\item Each iteration $j$ preoceeds with in a direction $v_j \sim \mathcal{U}({-1, 1})$ taking $2^j$ Leapfrog steps.
				\pause
				\item The iteration stops when the trajectory starts to double back on itself
				\pause
				\item or when the error in the simulation becomes too large, $\mathcal{L}(\theta)-\frac{1}{2}r\cdot r - \log{u} < -\Delta_{max}$.
			\end{itemize}
			\pause
			The doubling process defines the distribution $p(\mathcal{B} | \theta^t, r, u, \epsilon)$.

		\end{column}
		
        \begin{column}<1->{0.47\textwidth}
		\onslide<1->

\begin{frame}{Detailed NUTS algorithm}
	
	\begin{columns}

        \begin{column}{0.55\textwidth}
        \small

			\begin{itemize}
				\item NUTS algorithm repeatedly calls the BuildTree function, returning left and right endpoints of the trajectory, and all visited states $\mathcal{C'}$.
				\pause
				\item If one of the two stepping conditions is met, the algorithm terminates and the new candidates $\mathcal{C'}$ are rejected.
				\pause
				\item Otherwise, the new candidates $\mathcal{C'}$ are added to the set of candidate points $\mathcal{C}$.
				\pause
				\item If a global U turn condition is met, the doubling process is terminated.
				\pause
				\item The new position-momentum state $(\theta^{t+1}, r)$ is sampled from the set of candidate points $\mathcal{C}$.
			\end{itemize}
			

		\end{column}

\begin{frame}{Dual Averaging for $\epsilon$}
	

	\small
	$\epsilon$ is set using stochastic optimization with vanishing adaptation.
	\pause

	\begin{columns}

		
        \begin{column}{0.5\textwidth}
        \small
		
		
			\begin{itemize}
				\item Defining an expectation $h(\epsilon) = \mathbb{E}_{t}[H_t |x] = \lim_{T \to \infty} \frac{1}{T} \sum_{t=1}^T \mathbb{E}_{t}[H_t | x]$.
				\pause
				\item Using the desired acceptance probability $\delta$ to define the error $H_t = \delta - \alpha_t$, where $\alpha_t$ is the acceptance probability at iteration $t$.
				\pause
				\item Updating the parameter with the statistics $H_t$: $x_{t+1} \leftarrow \mu - \frac{\sqrt{t}}{\gamma}\frac{1}{t +t_0}\sum_{i=1}^{t}H_i$, $\bar{x}_{t+1} \leftarrow \bar{x}_t + \eta_t x_{t+1} + (1 - \eta_t )\bar{x}_t$ with $\eta_t = t^{-k}$. 
			\end{itemize}
			

		\end{column}
		
        \begin{column}<1->{0.5\textwidth}
		\onslide<1->
            \centering
			\begin{tikzpicture}
				\node[opacity=0.75] {
            		\includegraphics[width=1.1\textwidth]{images/immagini_leonardo/Dual_Averaging.png} 
				};
			\end{tikzpicture} \\
        \end{column}    
    
	
	\end{columns}
	\pause
	\small
	\vspace{1pt}
	$\mu=\log{10\epsilon_1}$ is a freely chosen point, $\gamma = 0.05$ is a free parameter for the speed of convergence, and $t_0=10$ is a free parameter that stabilizes the initial iterations and $k=0.75 \in (0.5, 1]$.

\end{frame}

\begin{frame}{Heuristic function for $\epsilon$}
	

	\small
	Convergence of Dual Averaging for $\epsilon$ is faster starting from a reasonable setting of parameters.
	\pause

	\begin{columns}

		
        \begin{column}{0.5\textwidth}
        \small
		
		
			\begin{itemize}
				\item This heuristic doubles or halves the value of $\epsilon_1$ until the acceptance probability of the Langevin (Leapfrog) proposal crosses 0.5.
				\pause
				\item NUTS does not have a single acceptance probability, so we use an average HMC acceptance probability over the final doubling iteration states.
			\end{itemize}
			

		\end{column}
		
        \begin{column}<1->{0.55\textwidth}
		\onslide<1->
            \centering
			\begin{tikzpicture}
				\node[opacity=0.75] {
            		\includegraphics[width=1\textwidth]{images/immagini_leonardo/Reasonable_Epsilon.png} 
				};
			\end{tikzpicture} \\
			\pause
			\vspace{3pt}
			$H^{NUTS}=\frac{1}{|\mathcal{B}_t^{f}|}\sum_{\theta, r \in \mathcal{B}_t^{f}} \min{1, \frac{\pi(\theta, r)}{\pi(\theta_{t-1}, r_{t}^{0})}}$.
		\end{column}    
    
	
	\end{columns}
	\small
	\vspace{3pt}
	\pause
	We can use Dual Averaging with $H_t = \delta - H^{NUTS}$ and $x_t = \log{\epsilon_t}$ to coerce the average of $H^{NUTS}$ to $\delta$. \\
	\pause
	\vspace{3pt}
	Now the algorithm only requires an acceptance probability $\delta$ as input, and a number of iterations $M^{adapt}$ for the adaptation phase.

\end{frame}

\begin{frame}{Effective Sample Size (ESS) (1)}
	
	MCMC algorithms generate correlated samples $\theta_1, \theta_2, \dots, \theta_M \sim p(\theta)$. Because these samples are dependent:
	\[
	1000 \text{ MCMC samples} \alert{\neq} 1000 \text{ independent samples}
	\]
	
	\vspace{0.5em}
	
	\textbf{ESS} estimates how many \alert{independent samples} contain the same information as the $M$ autocorrelated samples:
	
	\[
	\text{ESS} = \frac{M}{1 + 2 \sum_{s=1}^{M-1} \left(1 - \frac{s}{M}\right) \rho_s (f(\theta))}
	\]
	
	\vspace{0.5em}
	
	{\small
		\begin{itemize}
			\item $M$: total number of generated samples
			\item $\rho_s$: autocorrelation of the chain at lag $s$
			\item $\left(1 - \frac{s}{M}\right)$: finite-sample weight factor
		\end{itemize}
	}
	
\end{frame}


% =========================================================
% SLIDE 2: ESS Depends on the Function f(\theta)
% =========================================================
\begin{frame}{Effective Sample Size (ESS) (2)}
	
	ESS depends on the specific expectation \alert{$f(\theta)$} being estimated:
	
	\vspace{0.3em}
	
	To evaluate samplers on high-dimensional targets (e.g., 250D Gaussian), the authors of the NUTS paper compute two separate ESS values for each dimension $j$:
	
	\vspace{0.3em}
	
	\begin{itemize}
		\item \textbf{First Moment (Mean):} $f(\theta) = \theta_j \implies \text{ESS}_j^{(\text{mean})}$
		\begin{itemize}
			\item Measures efficiency in estimating the central location.
		\end{itemize}
		\vspace{0.2em}
		\item \textbf{Second Central Moment (Variance):} $f(\theta) = (\theta_j - \mu_{\text{true}})^2 \implies \text{ESS}_j^{(\text{variance})}$
		\begin{itemize}
			\item Measures efficiency in capturing uncertainty and scale.
			\item Requires a long baseline run (e.g., 50,000 samples) to obtain $\mu_{\text{true}}$.
		\end{itemize}
	\end{itemize}
	
\end{frame}

% =========================================================
% SLIDE 3 ESS
% =========================================================


\begin{frame}{Modern ESS diagnostics}
	
	Modern tools (PyMC / ArviZ) default to rank-normalized ESS (Vehtari et al., 2021):
	\begin{itemize}
		\item \textbf{Bulk ESS} (\texttt{"bulk"}): Rank-based measure of sampling efficiency in the center.
		\item \textbf{Tail ESS} (\texttt{"tail"}): Efficiency for extreme quantiles ($5\%$ and $95\%$).
	\end{itemize}
	
	\vspace{0.8em}
	
	\alert{Our Methodology:} 
	Despite modern defaults, in this project we use \textbf{$\text{ESS}^{(\text{mean})}$} and \textbf{$\text{ESS}^{(\text{variance})}$} to remain strictly coherent with the benchmark in the original NUTS paper (Hoffman \& Gelman, 2014).
	
\end{frame}



% =========================================================
% SLIDE 4 ESS
% =========================================================
\begin{frame}{Aggregating ESS}
	
	Since ESS is computed per scalar component, we compute a single ESS separately for each dimension. We then aggregate:
	
	% \vspace{0.5em}
	
	\begin{enumerate}
		
		\item \textbf{Joint Multi-Chain Calculation:}
		All parallel chains (e.g., 4 chains) are fed into the estimator simultaneously to penalize poor mixing, yielding a value of both $\text{ESS}_j^{(\text{mean})}$ and $\text{ESS}_j^{(\text{variance})}$ for each dimension
		
		\item \textbf{Conservative Dimension Reduction:}
		\begin{itemize}
			\item \textit{Per dim. $j$:} Take the minimum of mean and variance ESS:
			\[
			\text{ESS}_j = \min\left(\text{ESS}_j^{(\text{mean})}, \, \text{ESS}_j^{(\text{variance})}\right)
			\]
			\item \textit{Across all dimensions:} Take the worst-case dimension:
			\[
			\text{ESS}_{\text{final}} = \min_{j \in \{1, \dots, 250\}} \text{ESS}_j
			\]
		\end{itemize}
	\end{enumerate}
	
	\alert{Result:} A single, conservative metric reflecting the worst-explored feature of the distribution.
	
\end{frame}

	The target precision matrix $A = \Sigma^{-1}$ dictates the geometry of the target space:

	\end{itemize}

\section{Logistic Regression on German Credit Data}
\begin{frame}
	\begin{center}
		{\Huge \textbf{Logistic Regression on German Credit Data}}
	\end{center}
\end{frame}

\begin{frame}{The Dataset}
	\begin{enumerate}
		\item Original dataset contains both numerical (e.g. Age, credit amount, ...) and categorical data (e.g. Sex, Civil status, ...) $\rightarrow$ we used the numerical version provided by Strathclyde University.
		\item 1000 individuals with 24 features and a classifier for "good" (+1) and "bad" (-1) creditors.
	\end{enumerate}
	\begin{center}
		\textbf{Our goal:} sample from the target distribution of the posterior. 
	\end{center}
\end{frame}

\begin{frame}{The Model}
	\begin{center}
		\textbf{Bayesian Logistic Regression model.}
	\end{center}
	
	The target distribution is $p(\alpha, \beta \mid X, y) \propto p(y|X, \alpha, \beta) \, p(\alpha) \, p(\beta)$
	\begin{equation*}
		\propto \exp\left(
		-\sum_{i}\log\!\left(1+\exp\!\left\{-y_i\left(\alpha + x_i \cdot \beta\right)\right\}\right)
		-\frac{\alpha^2}{2\sigma^2}
		-\frac{1}{2\sigma^2}\,\beta \cdot \beta
		\right)
	\end{equation*}
	
	where:
	\begin{itemize}
		\item $x_i$ is a 24-dimensional vector of numerical predictors associated with a customer $i$.
		\item $y_i$ is $+1$ if the customer should receive credit, $-1$ otherwise.
		\item $\alpha$ is a scalar intercept term.
		\item $\beta$ is a vector of 24 regression coefficients.
		\item All predictors are normalized to have zero mean and unit variance.
		\item $\alpha$ and each element of $\beta$ are given given weak zero-mean normal priors with a \textbf{fixed} variance $\sigma^2 = 100$.
		
	\end{itemize}
	
\end{frame}

\begin{frame}{The Model: Likelihood}
	We assume that the probability of a customer being "good" depends on a linear combination of the features: 
	\begin{gather*}
		\eta_i = \alpha + x_i \beta \\
		p(y_i = 1 \mid x_i, \alpha, \beta) = \sigma(\eta_i)
	\end{gather*}
	where $\sigma(\eta_i) = \frac{1}{1 + e^{-\eta_i}}$ is the sigmoid logistic function.
	
	Therefore, because $y = \{-1, +1\}$
	$$
	\begin{cases}
		p(y_i = 1 | x_i) = \sigma(\eta_i) \\
		p(y_i = -1 | x_i) = 1 - \sigma(\eta_i) = \frac{e^{-\eta_i}}{1+e^{-\eta_i}} = \frac{1}{1 + e^{+\eta_i}}\\
	\end{cases}
	$$
	
	And we obtain the compact form:
	\begin{equation*}
		p(y_i | x_i) = \sigma(y_i \eta_i) = \frac{1}{1 + e^{-y_i \eta_i}}
	\end{equation*}
\end{frame}

\section{Hierarchical Logistic Regression on German Credit Data}
\begin{frame}
	\begin{center}
		{\Huge \textbf{Hierarchical Logistic Regression on German Credit Data}}
	\end{center}
\end{frame}
\begin{frame}{Changes from previous model}
	\begin{enumerate}
		\item We expand the predictor vectors by including two-way interactions ($x_i \cdot x_j$, $i \neq j$), resulting in $\binom{24}{2} = 276 + 24 =$ 300-dimensional vectors of predictors \textbf{x} and a 300-dimensional vector of coefficients \textbf{$\beta$}.
		\item The variance parameter in
		the prior on $\alpha$ and $\beta$ is given an exponential prior and estimated as well.
		\item  We set $\lambda = 0.01$, yielding a weak exponential prior distribution on $\sigma^2$,  whose mean and standard deviation are 100.
	\end{enumerate}
	
	\begin{center}
		$\Downarrow$
		
		These changes make for a more challenging problem: the posterior is in higher dimensions and the variance term $\sigma^2$ interacts strongly with the remaining 301 variables.
	\end{center}
\end{frame}

\begin{frame}{The paper's posterior}
	$p(\alpha, \beta , \sigma^2 \mid X, y) \propto  p(y|X, \alpha, \beta) \, p(\beta| \sigma^2) p(\alpha|\sigma^2) p(\sigma^2) \, \propto
	$
	$$
	\exp\left(
	-\sum_{i}\log\!\left(1+\exp\!\left\{-y_i x_i \cdot \beta\right\}\right)
	-\frac{\alpha^2}{2\sigma^2}
	-\frac{1}{2\sigma^2}\,\beta \cdot \beta - \frac{N}{2}\log{\sigma^2} - \lambda \sigma^2
	\right)
	$$
	where N = 1000 is the number of customers.
	
	\begin{center}
		{\Large \textbf{However...}}
		
		\Large
		that should instead be $\mathbf{\frac{D+1}{2}}$ where D is the dimension of $\beta$ (300).
		
	\end{center}
	
	\begin{center}
		(Also the intercept term $\alpha$ is missing in the (log)-likelihood)
	\end{center}
\end{frame}

\begin{frame}{Motivation}
	Now the variance $\sigma^2$ on $\alpha$ and $\beta$ is inferred from data, therefore in the log-posterior for these parameters we get:
	\begin{align*}
		\log p(\alpha \mid \sigma^2) &= -\frac{1}{2}\log\left(2\pi\sigma^2\right) - \frac{\alpha^2}{2\sigma^2} \\
		\log p(\beta \mid \sigma^2) &= -\frac{D}{2}\log\left(2\pi\sigma^2\right) - \frac{\beta^T \beta}{2\sigma^2}
	\end{align*}
	
	So their total contribution to the log-posterior is:
	\begin{equation*}
		-\frac{\alpha^2 + \beta^T \cdot \beta}{2\sigma^2} -\frac{D+1}{2}\log(\sigma^2)
	\end{equation*}
	
	Therefore, the log-posterior we sampled from is proportional to:
	\begin{equation*}
		-\sum_{i}\log\!\left(1+\exp\!\left\{-y_i\left(\alpha + x_i \cdot \beta\right)\right\}\right)
		-\frac{\alpha^2}{2\sigma^2}
		-\frac{\beta^T\beta}{2\sigma^2}\, - \frac{D+1}{2}\log{\sigma^2} - \lambda \sigma^2
	\end{equation*}
	
\end{frame}
